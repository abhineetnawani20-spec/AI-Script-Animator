    <script>
        function generatePreview() {
            const script = document.getElementById('scriptInput').value.trim();
            const style = document.getElementById('styleSelect').value;
            const previewArea = document.getElementById('previewArea');
            const statusIndicator = document.getElementById('statusIndicator');

            if (script === "") {
                alert("Please enter your script first!");
                return;
            }

            statusIndicator.innerText = "Rendering...";
            statusIndicator.style.color = "#3b82f6";

            setTimeout(() => {
                previewArea.className = "preview-box";

                let styleTitle = "";
                if (style === 'anime') {
                    previewArea.classList.add('style-anime');
                    styleTitle = "Anime Animation Keyframe";
                } else if (style === 'cartoon') {
                    previewArea.classList.add('style-cartoon');
                    styleTitle = "Cartoon Motion Sequence";
                } else if (style === 'note') {
                    previewArea.classList.add('style-note');
                    styleTitle = "Whiteboard Sketch Note";
                } else if (style === 'notyourtype') {
                    previewArea.classList.add('style-notyourtype');
                    styleTitle = "Not Your Type Infographic Sequence";
                }

                previewArea.innerHTML = `
                    <div class="output-content">
                        <div style="font-size: 11px; color: var(--text-muted); margin-bottom: 6px; text-transform: uppercase; letter-spacing: 0.5px;">[Mode: ${styleTitle}]</div>
                        <div style="font-weight: 500;">"${script}"</div>
                    </div>
                `;

                statusIndicator.innerText = "Completed";
                statusIndicator.style.color = "#10b981";
            }, 300);
        }

        // PWA Installation Trigger Logic
        let deferredPrompt;
        const installBtn = document.getElementById('installBtn');

        window.addEventListener('beforeinstallprompt', (e) => {
            e.preventDefault();
            deferredPrompt = e;
            installBtn.style.display = 'block';
        });

        installBtn.addEventListener('click', async () => {
            if (deferredPrompt) {
                deferredPrompt.prompt();
                const { outcome } = await deferredPrompt.userChoice;
                if (outcome === 'accepted') {
                    console.log('User accepted the install prompt');
                }
                deferredPrompt = null;
                installBtn.style.display = 'none';
            }
        });

        window.addEventListener('appinstalled', () => {
            installBtn.style.display = 'none';
            console.log('PWA was installed successfully');
        });

        // Register Service Worker
        if ('serviceWorker' in navigator) {
            window.addEventListener('load', () => {
                navigator.serviceWorker.register('sw.js')
                    .then((reg) => console.log('Service Worker registered successfully', reg))
                    .catch((err) => console.log('Service Worker registration failed', err));
            });
        }
    </script>
